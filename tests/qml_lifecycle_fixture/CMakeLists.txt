cmake_minimum_required(VERSION 3.28)
project(GenycasterQmlLifecycleFixture LANGUAGES CXX)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Qml Quick)
set(root "${CMAKE_CURRENT_LIST_DIR}/../..")
add_library(teardown STATIC)
target_sources(teardown PUBLIC FILE_SET CXX_MODULES BASE_DIRS "${root}/ui/app/bootstrap" FILES "${root}/ui/app/bootstrap/teardown.cppm" PRIVATE "${root}/ui/app/bootstrap/teardown.cpp")
target_include_directories(teardown PRIVATE "${root}/ui/app/runtime")
target_compile_features(teardown PUBLIC cxx_std_26)
target_link_libraries(teardown PUBLIC Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
add_executable(bootstrap_teardown_fixture "${root}/tests/ui/bootstrap_teardown_tests.cpp")
target_link_libraries(bootstrap_teardown_fixture PRIVATE teardown)
enable_testing()
add_test(NAME bootstrap_teardown COMMAND bootstrap_teardown_fixture)
set_tests_properties(bootstrap_teardown PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_FATAL_WARNINGS=1" TIMEOUT 15)
find_package(Python3 QUIET COMPONENTS Interpreter)
find_program(QML_TEST_RUNNER NAMES qmltestrunner HINTS "${Qt6_DIR}/../../../bin" NO_DEFAULT_PATH)
if(Python3_Interpreter_FOUND AND QML_TEST_RUNNER)
  add_test(NAME supporting_preview_lifecycle COMMAND "${Python3_EXECUTABLE}" "${root}/tests/ui/supporting_preview_lifecycle_runtime.py")
  set_tests_properties(supporting_preview_lifecycle PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QML_TEST_RUNNER=${QML_TEST_RUNNER}"
    TIMEOUT 40)
endif()
